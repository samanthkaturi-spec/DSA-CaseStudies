Road Network Construction Using Kruskal's Algorithm
class Graph:
    def __init__(self, v):
        self.V = v
        self.graph = []

    def addEdge(self, u, v, w):
        self.graph.append([u, v, w])

    def kruskal(self):
        self.graph = sorted(self.graph, key=lambda item: item[2])

        result = []
        parent = list(range(self.V))

        def find(i):
            while parent[i] != i:
                i = parent[i]
            return i

        for u, v, w in self.graph:
            x = find(u)
            y = find(v)

            if x != y:
                result.append((u, v, w))
                parent[x] = y

        print("Minimum Spanning Tree:")
        total_cost = 0

        for u, v, w in result:
            print(f"{u} -- {v} = {w}")
            total_cost += w

        print("Total Cost =", total_cost)

# Driver Code
g = Graph(4)

g.addEdge(0, 1, 10)
g.addEdge(0, 2, 6)
g.addEdge(0, 3, 5)
g.addEdge(1, 3, 15)
g.addEdge(2, 3, 4)

g.kruskal()
